class Solution
{
public:
    int smallestDivisor(vector<int> &nums, int threshold)
    {
        int f = 1, l = *max_element(nums.begin(), nums.end());
        while (f <= l)
        {
            int mid = f + ((l - f) / 2);
            int ans = 0;
            for (int i = 0; i < nums.size(); i++)
            {
                ans = ans + ceil((double)nums[i] / mid);
            }
            if (ans > threshold)
            {
                f = mid + 1;
            }
            else
                l = mid - 1;
        }
        return f;
    }
};
